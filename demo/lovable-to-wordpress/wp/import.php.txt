$b = array(
	'p'  => function ( $text, $class = '' ) {
		return $class ? "<!-- wp:paragraph {\"className\":\"$class\"} -->\n<p class=\"$class\">$text</p>\n<!-- /wp:paragraph -->\n"
			: "<!-- wp:paragraph -->\n<p>$text</p>\n<!-- /wp:paragraph -->\n";
	},
	'h'  => function ( $text, $level = 2 ) {
		$attr = 2 === $level ? '' : " {\"level\":$level}";
		return "<!-- wp:heading$attr -->\n<h$level class=\"wp-block-heading\">$text</h$level>\n<!-- /wp:heading -->\n";
	},
	'g'  => function ( $class, $inner, $tag = 'div' ) {
		$attr = 'div' === $tag ? "{\"className\":\"$class\"}" : "{\"tagName\":\"$tag\",\"className\":\"$class\"}";
		return "<!-- wp:group $attr -->\n<$tag class=\"wp-block-group $class\">$inner</$tag>\n<!-- /wp:group -->\n";
	},
	'btn' => function ( $list ) {
		$out = '';
		foreach ( $list as $x ) {
			list( $text, $url, $style ) = $x;
			$out .= "<!-- wp:button {\"className\":\"$style\"} -->\n<div class=\"wp-block-button $style\"><a class=\"wp-block-button__link wp-element-button\" href=\"" . esc_url( $url ) . "\">$text</a></div>\n<!-- /wp:button -->\n";
		}
		return "<!-- wp:buttons -->\n<div class=\"wp-block-buttons\">$out</div>\n<!-- /wp:buttons -->\n";
	},
	'sc' => function ( $code ) {
		return "<!-- wp:shortcode -->\n$code\n<!-- /wp:shortcode -->\n";
	},
	'qa' => function ( $q, $a ) {
		return "<!-- wp:details -->\n<details class=\"wp-block-details\"><summary>$q</summary><!-- wp:paragraph -->\n<p>$a</p>\n<!-- /wp:paragraph --></details>\n<!-- /wp:details -->\n";
	},
);
